\chapterbanner{Chapitre 5 --- Stabilit\'e \& Marges}{ch5red}

\begin{tcolorbox}[colback=ch5red!5,colframe=ch5red!80!black,title={Stabilit\'e BIBO}]
\tightmath{
\text{Stable (BIBO)} \Leftrightarrow \textbf{tous les p\^oles} : \text{Re}(s_i) < 0
}
\begin{itemize}
  \item Marginalement stable : p\^ole(s) simple(s) sur axe $j\omega$
  \item Instable : au moins un p\^ole avec Re$(s_i) > 0$
\end{itemize}
\vars{\'Etude par la r\'eponse impulsionnelle (\S5.2, p.165) : mode ap\'eriodique $e^{p_i t}$ born\'e ssi $p_i<0$ ; mode oscillatoire $e^{-\delta t}\sin(\omega_0 t)$ born\'e ssi $\delta>0$.}
\piege{En pratique on ne calcule pas les p\^oles : on juge la stabilit\'e \textbf{BF} sur le lieu de $G_o(\jw)$ (crit\`ere du revers) ou sur le Bode de $G_o$ (marges).}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch5red!5,colframe=ch5red!80!black,title={Stabilit\'e en boucle ferm\'ee}]
\'Equation caract\'eristique :
\tightmath{
\boxed{1 + G_o(s) = 0} \quad G_o = G_c\,G_a
}
BF stable $\Leftrightarrow$ toutes les racines de $1 + G_o(s) = 0$ ont Re $< 0$.
\end{tcolorbox}

\begin{tcolorbox}[colback=ch5red!5,colframe=ch5red!80!black,title={Distance critique $d_{\min}$}]
Distance minimale entre le lieu de Nyquist de $G_o(\jw)$ et le point critique $-1$ :
\tightmath{
\boxed{d_{\min} = \min_{\omega} |1 + G_o(\jw)|}
}
\vars{Chiffre le \textbf{degr\'e de stabilit\'e}. Plus $d_{\min}$ est grand, plus la BF est robuste. En pratique on l'\'evalue indirectement par les marges $\varphi_m$ et $A_m$.}
\tightmath{
\|S\|_\infty = \frac{1}{d_{\min}} = \max_\omega\Big|\frac{1}{1+G_o(\jw)}\Big|
}
\piege{$d_{\min}$ \textbf{petit} $\Rightarrow$ pic de r\'esonance \'elev\'e en BF $\Rightarrow$ fort d\'epassement.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch5red!5,colframe=ch5red!80!black,title={Marge de gain ($A_m$)}]
$\omega_\pi$ = \textbf{croisement de phase} : $\angle G_o(\jw_\pi) = -180°$.
\tightmath{
\boxed{A_m = \frac{1}{|G_o(\jw_\pi)|}} \quad \boxed{A_{m,\text{dB}} = -|G_o(\jw_\pi)|_{\text{dB}}}
}
\vars{Sur Bode : $A_{m,\text{dB}}$ = distance module $\leftrightarrow$ 0~dB, \textbf{lue \`a $\omega_\pi$}. Stable si courbe \textbf{en dessous} de 0~dB. Typ. : $A_m \geq 6$--$10$ dB.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch5red!5,colframe=ch5red!80!black,title={Marge de phase ($\varphi_m$)}]
$\omega_{co}$ = \textbf{croisement de gain} : $|G_o(\jw_{co})| = 1$ (0~dB).
\tightmath{
\boxed{\varphi_m = 180° + \angle G_o(\jw_{co})}
}
\vars{Sur Bode : $\varphi_m$ = distance phase $\leftrightarrow$ $-180°$, \textbf{lue \`a $\omega_{co}$}. Stable si phase \textbf{au-dessus} de $-180°$. Objectif : $30° < \varphi_m < 60°$.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch5red!5,colframe=ch5red!80!black,title={Relations approx. $\varphi_m$ -- $\zeta$ -- $D_\%$}]
\tightmath{
\boxed{\zeta \approx \frac{\varphi_m}{100°}} \quad (\text{pour } 20° < \varphi_m < 70°)
}
\vars{$\varphi_m = 45° \Rightarrow \zeta \approx 0.45 \Rightarrow D_\% \approx 20\%$. $\varphi_m = 65° \Rightarrow \zeta \approx 0.65 \Rightarrow D_\% \approx 7\%$.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch5red!5,colframe=ch5red!80!black,title={Approximations BF depuis Bode}]
\textbf{$T_{\text{r\'eg}}$ en BF} (approx. demi-p\'eriode \`a $\omega_{co}$) :
\tightmath{
\boxed{T_{\text{r\'eg}} \approx \frac{\pi}{\omega_{co}}}
}
\textbf{Retard pur marginalisant la BF} (consume $\varphi_m$ : $\omega_{co} T_r = \varphi_{m,\text{rad}}$) :
\tightmath{
\boxed{T_r^{\max} = \frac{\varphi_{m,\text{rad}}}{\omega_{co}} = \frac{\varphi_m° \cdot \pi}{180° \cdot \omega_{co}}}
}
\piege{$\varphi_m$ en \textbf{radians} : $\varphi_{m,\text{rad}} = \varphi_m° / 57.3$. Ex. : $\varphi_m{=}40°$, $\omega_{co}{=}600$ $\Rightarrow$ $T_r^{\max} = \frac{40\pi}{180\cdot600} \approx 1.2\,\text{ms}$.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch5red!5,colframe=ch5red!80!black,title={Crit\`ere du revers (Nyquist simplifi\'e)}]
\textbf{\'Enonc\'e (si $G_o$ stable en BO)} : le syst\`eme en BF est stable $\Leftrightarrow$ en parcourant le lieu de $G_o(\jw)$ dans le sens des $\omega$ croissants, le point critique $-1$ reste \textbf{\`a gauche} du lieu.
\begin{itemize}
  \item \'Equivalent : le lieu \textbf{n'entoure pas} $(-1,0)$
  \item Le croisement de l'axe r\'eel n\'egatif se fait \textbf{entre} $-1$ et $0$
\end{itemize}
\piege{Valable uniquement si la \textbf{boucle ouverte est stable}. Si $G_o$ a des p\^oles instables, utiliser le crit\`ere de Nyquist complet (comptage d'encerclements).}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch5red!5,colframe=ch5red!80!black,title={M\'ethode de Bode : marche \`a suivre}]
Synth\`ese d'un r\'egulateur pour imposer $\varphi_m$ et la rapidit\'e ($\omega_{co}$) :
\begin{enumerate}
  \item Choisir $\omega_{co}$ d\'esir\'ee (rapidit\'e : $T_{\text{r\'eg}}\approx\pi/\omega_{co}$)
  \item Choisir $\varphi_m$ d\'esir\'ee (amortissement : $\zeta\approx\varphi_m/100$)
  \item Tracer le Bode de $G_a$ (syst\`eme \`a r\'egler)
  \item Choisir la structure du r\'egulateur $G_c$ (P, PI, PD, PID\dots)
  \item \textbf{Phase} : \`a $\omega_{co}$, imposer $\angle G_o(\jw_{co}) = -180° + \varphi_m$ $\Rightarrow$ fixe les constantes de temps de $G_c$
  \item \textbf{Gain} : ajuster $K_p$ pour que $|G_o(\jw_{co})| = 1$ (0~dB)
  \item V\'erifier $A_m$ \`a $\omega_\pi$ et la pr\'ecision (gain permanent)
\end{enumerate}
\astuce{Compromis : $\omega_{co}$ grande $\Rightarrow$ rapide ; $\varphi_m$ grande $\Rightarrow$ peu de d\'epassement. Zone usuelle $\varphi_m\in[35°,60°]$, $A_m\ge 6$ dB.}
\end{tcolorbox}
