\chapterbanner{Chapitre 6 --- R\'egulateurs \& Boucle ferm\'ee}{ch6teal}

\begin{tcolorbox}[colback=ch6teal!4,colframe=ch6teal!80!black,title={Transferts en boucle ferm\'ee}]
Avec $L(s) = G_r(s) \cdot G(s)$ (transfert de boucle), r\'etroaction unitaire :
\tightmath{
\boxed{G_{yw} = \frac{L}{1{+}L}} \quad \boxed{G_{ew} = \frac{1}{1{+}L}} \quad \boxed{G_{yv} = \frac{G_p}{1{+}L}}
}
\tightmath{
S(s) + T(s) = 1 \quad (S = G_{ew},\ T = G_{yw})
}
\piege{Le d\'enom. $1{+}L(s)$ est commun \`a \textbf{tous} les transferts BF. \'Eq. caract. : $1{+}L(s){=}0$.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!2,colframe=ch6teal!80!black,title={Erreur permanente : constantes \& tableau}]
$K_p = \lim_{s\to 0} L(s)$ (position), $K_v = \lim_{s\to 0} sL(s)$ (vitesse), $K_a = \lim_{s\to 0} s^2 L(s)$ (acc\'el.)
\begin{center}
\renewcommand{\arraystretch}{1.1}
\scriptsize
\begin{tabular}{@{}lccc@{}}
\toprule
Entr\'ee $w(t)$ & Type 0 & Type 1 & Type 2 \\
\midrule
\'Echelon $A$ & $\frac{A}{1{+}K_p}$ & $0$ & $0$ \\[3pt]
Rampe $At$ & $\infty$ & $\frac{A}{K_v}$ & $0$ \\[3pt]
Parabole $\frac{A}{2}t^2$ & $\infty$ & $\infty$ & $\frac{A}{K_a}$ \\
\bottomrule
\end{tabular}
\end{center}
\textbf{Poursuite} ($v{=}0$) : $e_\infty = \lim_{s\to 0} \frac{1}{1{+}L(s)} \cdot \frac{A}{s} \cdot s = \frac{A}{1{+}K_{BO}}$

\textbf{Maintien} ($w{=}0$) : $e_\infty = -\lim_{s\to 0} \frac{G_p(s)}{1{+}L(s)}$
\piege{Type $\alpha$ \'elev\'e $\Rightarrow$ meilleure pr\'ecision, mais plus difficile \`a stabiliser.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!4,colframe=ch6teal!80!black,title={R\'egulateur P (Proportionnel)}]
\tightmath{
\boxed{G_r(s) = K_p}
}
\begin{itemize}
  \item R\'eduit $t_r$ (augmente $\omega_c$), augmente $D_\%$ ($MP \downarrow$)
  \item R\'eduit mais \textbf{ne supprime pas} $e_\infty$ (type 0)
  \item \textbf{Dilemme} : $K_p \uparrow \Rightarrow e_\infty \downarrow$ mais stabilit\'e $\downarrow$
\end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!2,colframe=ch6teal!80!black,title={R\'egulateur PI}]
\tightmath{
\boxed{G_r(s) = K_p\!\left(1 + \frac{1}{T_i s}\right) = K_p\,\frac{1 + T_i s}{T_i s}}
}
\begin{itemize}
  \item Ajoute int\'egrateur $\Rightarrow$ type $+1$ $\Rightarrow$ \textbf{supprime $e_\infty$} (\'echelon)
  \item Z\'ero en $s = -1/T_i$ : compenser p\^ole dominant
  \item Diminue $MP$ (attention stabilit\'e)
\end{itemize}
\astuce{\textbf{R\'eglage PI} : 1. $T_i = \tau_{\text{proc}}$ (annule p\^ole dominant). 2. R\'egler $K_p$ pour $MP$ souhait\'ee. 3. V\'erifier $\omega_c$ et $MG$.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!4,colframe=ch6teal!80!black,title={R\'egulateur PD \& PID}]
\textbf{PD} : $\boxed{G_r(s) = K_p(1 + T_d s)}$ \quad \textbf{PD r\'ealisable} :
\tightmath{
\boxed{G_r = K_p\!\left(1 + \frac{sT_d}{1{+}\alpha s T_d}\right) = K_p\,\frac{1{+}(1{+}\alpha)T_d s}{1{+}\alpha T_d s}}
}
\begin{itemize}
  \item Z\'ero \`a $\omega_z = \frac{1}{(1{+}\alpha)T_d}$, p\^ole \`a $\omega_p = \frac{1}{\alpha T_d}$, gain BF $= K_p$
  \item Am\'eliore amortissement, r\'eduit $D_\%$, augmente $MP$. Amplifie bruit HF.
\end{itemize}
\vars{\textbf{Identification depuis Bode} : $K_p = 10^{G_{\text{BF}}/20}$, puis z\'ero $\omega_z$ et p\^ole $\omega_p$ $\Rightarrow$ $T_d = 1/\omega_z - 1/\omega_p$, $\alpha = \omega_z/(\omega_p - \omega_z)$.}
\textbf{PID} :
\tightmath{
\boxed{G_r(s) = K_p\!\left(1 + \frac{1}{T_i s} + T_d s\right)}
}
Forme factoris\'ee : $G_r = K_p\frac{(1{+}T_1 s)(1{+}T_2 s)}{T_i s}$ avec $T_1{+}T_2 = T_i$, $T_1 T_2 = T_i T_d$.
\astuce{\textbf{PID par compensation} : placer les 2 z\'eros sur les 2 p\^oles dominants. $T_1 = -1/p_1$, $T_2 = -1/p_2$. R\'egler $K_p$ pour $\omega_c$ et $MP$.}
\end{tcolorbox}

\end{tabular}
\end{center}
\textbf{M\'ethode BF} (oscillation) : $K_u$ (gain critique), $T_u$ (p\'eriode osc.).
\begin{center}

\begin{tabular}{@{}lccc@{}}

\begin{tcolorbox}[colback=ch6teal!4,colframe=ch6teal!80!black,title={Correcteur \`a avance de phase (Lead)}]
\tightmath{
\boxed{C(s) = K_c \cdot \frac{1 + \tau s}{1 + \alpha\tau s}, \quad 0 < \alpha < 1}
}
\begin{itemize}
  \item Avance max : $\varphi_m = \arcsin\frac{1{-}\alpha}{1{+}\alpha}$ \;\`a\; $\omega_m = \frac{1}{\tau\sqrt{\alpha}}$
  \item Gain \`a $\omega_m$ : $+10\log(1/\alpha)$ dB
  \item Effet : augmente $MP$, augmente $\omega_c$
\end{itemize}
\textbf{Dimensionnement :}
\begin{enumerate}
  \item $\varphi_m = MP_{\text{souh}} - MP_0 + 5°$--$12°$
  \item $\alpha = \frac{1 - \sin\varphi_m}{1 + \sin\varphi_m}$
  \item Nouvelle $\omega_c$ : l\`a o\`u $|G_{BO}| = -10\log(1/\alpha)$ dB
  \item $\tau = 1/(\omega_c\sqrt{\alpha})$
  \item $K_c$ ajuste le gain si n\'ecessaire
\end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!2,colframe=ch6teal!80!black,title={Correcteur \`a retard de phase (Lag)}]
\tightmath{
\boxed{C(s) = K_c \cdot \frac{1 + \tau s}{1 + \beta\tau s}, \quad \beta > 1}
}
\begin{itemize}
  \item Multiplie $K_{BO}$ par $\beta$ $\Rightarrow$ meilleure pr\'ecision
  \item $\tau$ grand : cassure $1/(\beta\tau)$ bien en dessous de $\omega_c$
  \item Ne modifie pas (ou peu) $MP$
\end{itemize}
\vars{$\beta =$ gain suppl. n\'ecessaire. Placer $1/\tau \approx \omega_c/10$.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!2,colframe=ch6teal!80!black,title={Proc\'edures types d'examen}]
\textbf{Identifier $G(s)$ depuis Bode :} pente BF $\to$ type. Prolonger \`a $\omega{=}1 \to K$. Cassures $\to$ p\^oles/z\'eros. Pic $\to \zeta$.

\textbf{Erreur permanente :} identifier transfert BF ($G_{ew}$ ou $G_{yv}$). Th. val. finale. Ou tableau type/entr\'ee.

\textbf{Routh param\'etrique :} \'eq. caract. $1{+}KG_{BO}{=}0$. Tableau. 1\textsuperscript{\`ere} col. $> 0 \Rightarrow$ plage $K$.

\textbf{R\'eglage PI :} $T_i = \tau_{\text{proc}}$. Bode de $L(s)$. $K_p$ pour $MP$ souhait\'ee.

\textbf{Lead :} $\varphi_m \to \alpha \to \omega_c \to \tau$. Retracer Bode. V\'erifier $MP$, $MG$.
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!4,colframe=ch6teal!80!black,title={Placement des p\^oles : m\'ethode rapide}]
\'Eq. caract. BF : $1 + G_{BO}(s) = 0$. Mettre sous forme polynomiale, choisir $p_1,p_2,\dots$, former $(s-p_1)(s-p_2)\cdots$, puis identifier les coefficients pour trouver $K_p,T_i,T_d,\dots$
\tightmath{
\text{Ordre 2 : } (s-p_1)(s-p_2) = s^2 - (p_1+p_2)s + p_1p_2
}
\vars{Il faut assez de param\`etres r\'eglables pour imposer les coefficients voulus.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!4,colframe=ch6teal!80!black,title={Transferts BF avec capteur $G_m \neq 1$}]
\tightmath{
G_{yw} = \frac{G_r G_a G_p}{1 + G_r G_a G_p G_m} \quad L(s) = G_r G_a G_p G_m
}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!4,colframe=ch6teal!80!black,title={Retard, Pad\'e \& p\^ole dominant}]
\textbf{Retard} : $MP_{\text{avec}} = MP_{\text{sans}} - \omega_c T_r \cdot 180/\pi$.\\
\textbf{Pad\'e / d\'ephaseur} : $e^{-sT_r} \approx \frac{1 - T_r s/2}{1 + T_r s/2}$,\; $|G_d(j\omega)| \approx 1$.\\
\textbf{P\^ole dominant} : $t_{s5} \approx 3/|\text{Re}(s_{\text{dom}})|$, $t_{s2} \approx 4/|\text{Re}(s_{\text{dom}})|$.
\vars{Dominant si les autres p\^oles ont $|\text{Re}| \geq 5\times$ plus grande.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!4,colframe=ch6teal!80!black,title={Pi\`eges fr\'equents aux examens}]
\begin{itemize}
  \item Confondre $K$ et $20\log K$ (dB)
  \item BF : d\'enom. $1{+}L$, pas $1{-}L$
  \item $MG$ = distance \`a 0~dB lue \textbf{\`a $\omega_\pi$} (pas $\omega_c$)
  \item $MP$ = distance \`a $-180°$ lue \textbf{\`a $\omega_c$} (pas $\omega_\pi$)
  \item Confondre $\omega$ (rad/s) et $f$ (Hz) : $\omega = 2\pi f$
  \item PI : $T_i = \tau_{\text{proc}}$ pour annuler p\^ole dominant
  \item $e_\infty$ : toujours utiliser transfert \textbf{BF}, jamais BO
  \item Val. finale : ne pas appliquer si syst\`eme instable
  \item Type du syst\`eme = nb int\'egrateurs en \textbf{boucle ouverte}
  \item Bode retard : module inchang\'e, seule la phase chute
\end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!2,colframe=ch6teal!80!black,title={Check-list avant l'examen}]
\begin{enumerate}
  \item Passer entre les 3 formes de $G(s)$
  \item Tracer un Bode asymptotique rapidement
  \item Lire $K$, type, $\omega_c$, p\^oles/z\'eros sur Bode
  \item Calculer $MG$ et $MP$ sur Bode donn\'e
  \item Appliquer Routh-Hurwitz (+ param\'etrique)
  \item D\'ecomposer en fractions partielles
  \item Calculer $e_\infty$ pour \'echelon/rampe
  \item Dimensionner PI, PID, Lead ou Lag
  \item Appliquer Ziegler-Nichols (2 m\'ethodes)
  \item Th. valeur finale correctement
\end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!4,colframe=ch6teal!80!black,title={M\'ethode de conception BF (workflow complet)}]
\begin{enumerate}
  \item Fixer le cahier des charges : $e_\infty$, $D_\%$, $t_s$, robustesse (MP/MG)
  \item Choisir une structure : P, PI, PD, PID, Lead, Lag selon l'objectif principal
  \item Faire un premier dimensionnement analytique ($T_i$, $T_d$, $\alpha$, $\tau$, ...)
  \item Ajuster $K_p$ pour positionner $\omega_c$ et obtenir MP vis\'ee
  \item V\'erifier : BF stable, $e_\infty$ conforme, saturation actionneur acceptable
\end{enumerate}
\vars{Priorit\'e classique : pr\'ecision statique (I/Lag), puis dynamique (D/Lead), puis robustesse (MP, MG).}
\piege{Un r\'eglage qui satisfait seulement Bode sans contr\^ole de saturation peut \^etre inutilisable en pratique.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!2,colframe=ch6teal!80!black,title={Sensibilit\'e \& robustesse}]
\tightmath{
\boxed{S(s) = \frac{1}{1{+}L},\quad T(s) = \frac{L}{1{+}L},\quad S + T = 1}
}
\begin{itemize}
  \item $S$ : passe-haut $\Rightarrow$ att\'enue les perturbations BF ; $T$ : passe-bas
  \item Robustesse : $\|S\|_\infty \leq 2$ ($6$~dB) recommand\'e $\Leftrightarrow$ MP $\geq 30°$
  \item $\|S\|_\infty = 1/\text{dist}(G_{BO}(\jw),\,-1)_{\min}$
\end{itemize}
\piege{Minimiser $e_\infty$ et maximiser $MP$ sont des objectifs conflictuels : c'est le dilemme pr\'ecision-stabilit\'e.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!2,colframe=ch6teal!80!black,title={Calcul de $G(j\omega)$ : module et phase}]
Substituer $s = j\omega$ puis appliquer :
\tightmath{
|j\omega| = \omega,\quad |1{+}j\omega\tau| = \sqrt{1{+}(\omega\tau)^2},\quad \angle j\omega = 90°
}
\tightmath{
\angle(1{+}j\omega\tau) = \arctan(\omega\tau),\quad \angle\frac{A}{B} = \angle A - \angle B
}
\vars{Produit : $|AB|=|A||B|$, $\angle AB = \angle A + \angle B$. Gain r\'eel $K{>}0$ : $\angle K = 0°$. Si $K{<}0$ : $\angle = -180°$.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!4,colframe=ch6teal!80!black,title={R\'eponse sinuso\"idale BF : calcul num\'erique}]
Entr\'ee $w(t)=A\sin(\omega t)$ en BF, sortie en r\'egime perm. :
\tightmath{
y(t) = A\,|G_{BF}(j\omega)|\,\sin\!\bigl(\omega t + \angle G_{BF}(j\omega)\bigr)
}
\textbf{M\'ethode depuis Bode BO} :
\begin{enumerate}
  \item Lire $|G_o|$ et $\angle G_o$ \`a la puls. $\omega$ (pour $K_{p,0}$ du Bode)
  \item Corriger : $|G_o|_{\text{r\'eel}} = |G_o| \cdot K_p/K_{p,0}$, $\angle$ inchang\'e
  \item Former $G_o = M\angle\theta$ puis $1{+}G_o = (1+M\cos\theta)+jM\sin\theta$
  \item $|G_{BF}| = M/|1{+}G_o|$, \quad $\angle G_{BF} = \theta - \angle(1{+}G_o)$
\end{enumerate}
\piege{Si Re$(1{+}G_o) < 0$ : $\angle(1{+}G_o)$ est dans le demi-plan gauche, tenir compte du quadrant (atan2).}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!2,colframe=ch6teal!80!black,title={Erreur avec perturbation : recette compl\`ete}]
Superposition pour $W{=}A/s$ et $V{=}B/s$ simultan\'es :
\tightmath{
y_\infty = \underbrace{G_{yw}(0)\cdot A}_{\text{d\^u \`a }w} + \underbrace{G_{yv}(0)\cdot B}_{\text{d\^u \`a }v} \qquad e_\infty = A - y_\infty
}
\textbf{Type 0} : $G_{yw}(0){=}\frac{K_{BO}}{1{+}K_{BO}}$, $G_{yv}(0){=}\frac{G_p(0)}{1{+}K_{BO}}$ \\[1pt]
\textbf{Type 1} : $G_{yw}(0){=}1$, $G_{yv}(0){=}\lim_{s\to 0}\frac{G_p(s)}{1{+}L(s)}$ (fini)
\vars{Exemple type 1 : $L = 2K_p/[s(1{+}s)]$, $G_{yv}(0) = 1/(2K_p)$. Pour $V{=}0.5$ : $y_{v,\infty} = 0.5/(2K_p)$.}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!4,colframe=ch6teal!80!black,title={Gain critique $K_u$ par Routh (m\'ethode BO)}]
\begin{enumerate}
  \item Former $1 + K\,G_{BO}(s) = 0$, d\'evelopper en polyn\^ome en $K$
  \item Construire tableau de Routh avec coefficients d\'ependant de $K$
  \item $K_u$ = valeur qui annule un \'el\'ement de la 1\textsuperscript{\`ere} colonne
  \item Ligne nulle $\Rightarrow$ poly. auxiliaire $P(s)$ $\Rightarrow$ $s^2 + \omega_u^2 = 0$ $\Rightarrow$ $T_u = 2\pi/\omega_u$
\end{enumerate}
\astuce{Ziegler-Nichols BF utilise directement $K_u$ et $T_u$ (voir tableau ZN).}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!2,colframe=ch6teal!80!black,title={Lead : exemple de dimensionnement complet}]
\textbf{Donn\'ee} : $G(s) = \frac{10}{s(1{+}0.1s)}$, objectif $MP = 50°$. $MP_0 \approx 18°$.
\begin{enumerate}
  \item $\varphi_m = 50° - 18° + 8° = 40°$
  \item $\alpha = \dfrac{1-\sin40°}{1+\sin40°} \approx 0.217$
  \item Gain \`a apporter : $+10\log(1/0.217) \approx +6.6$ dB
  \item Chercher $\omega'$ tel que $|G(\jw')| = -6.6$ dB $\Rightarrow \omega_c' \approx 15$ rad/s
  \item $\tau = \dfrac{1}{\omega_c'\sqrt{\alpha}} \approx 0.143$ s
  \item $C(s) = \dfrac{1{+}0.143s}{1{+}0.031s}$ \quad(avec $\alpha\tau = 0.031$)
\end{enumerate}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!2,colframe=ch6teal!80!black,title={Correspondance BO $\to$ BF : r\`egles de pouce}]
\begin{itemize}
  \item $\omega_{-3\text{dB}}^{BF} \approx \omega_c^{BO}$ pour $30° < MP < 70°$
  \item $K_p \times N$ : $|G_{BO}|_{\text{dB}} \uparrow 20\log N$, phase \textbf{inchang\'ee}, $\omega_c$ se d\'eplace
  \item Bode de $G_{BF}$ : pic de r\'esonance \`a $\omega_r \approx \omega_c$ si MP petit
  \item $D_\% \approx 100\,e^{-\pi MP_{\text{rad}}}$ (approximation r\'egime sous-amorti)
  \item $T_{\text{r\'eg}} \approx \pi/\omega_c$, \quad $T_0^{BF} \approx 2\pi/\omega_c$
\end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!2,colframe=ch6teal!80!black,title={Points cl\'es du dimensionnement en pratique}]
\begin{itemize}
  \item Apr\`es ajout d'un correcteur, \textbf{retracer Bode complet} et reverifier $MP$, $MG$
  \item Retard non mod\'elis\'e $\Rightarrow$ prendre $10°$--$15°$ de marge suppl\'ementaire sur $MP$
  \item PD/Lead : gain HF $\uparrow$ $\Rightarrow$ amplifie bruit capteur, \`a limiter ($N$ ou $\alpha$ pas trop grand)
  \item PI : p\^ole en $s{=}0$ $\Rightarrow$ risque d'int\'egral windup si saturation actionneur
  \item Routh donne la plage de $K$ stable, pas la \textbf{marge de robustesse} : croiser avec Bode
  \item Crit\`eres pratiques : $MP \in [35°, 60°]$, $MG \geq 6$ dB, $T_r$ compatible avec $\omega_c$
\end{itemize}
\end{tcolorbox}

\begin{tcolorbox}[colback=ch6teal!4,colframe=ch6teal!80!black,title={R\'egulateur I pur \& double int\'egrateur en BO}]
\textbf{I pur} : $G_r(s) = K_i/s$ \\
$\Rightarrow$ ajoute un p\^ole en 0 (type $+1$), phase BO $-90°$ suppl. $\Rightarrow$ \textbf{r\'eduit $MP$}.\\[2pt]
\textbf{Type 2} ($1/s^2$ en BO) : \'elimine $e_\infty$ pour rampe, mais tr\`es difficile \`a stabiliser.\\[2pt]
\textbf{Bode de $1/s^2$} : pente $-40$ dB/d\'ec, phase $= -180°$ (constant). \\[2pt]
\piege{Pour type\,$\geq 2$ : la condition de Hurwitz seule ne suffit plus (ordre\,$\geq 3$). Appliquer Routh ou simuler.}

\begin{tcolorbox}[colback=ch6teal!2,colframe=ch6teal!80!black,title={Formes des p\^oles BF complexes connus}]
Polyn\^ome caract\'eristique $1{+}L(s)=0$ de type fr\'equent :
\tightmath{
s^2 + 2\zeta\omega_n s + \omega_n^2 = 0 \;\Rightarrow\; s = -\zeta\omega_n \pm j\omega_n\sqrt{1-\zeta^2}
}
\tightmath{
s^2 + bs + c = 0 \;\Rightarrow\; \omega_n = \sqrt{c},\quad \zeta = \frac{b}{2\sqrt{c}}
}
\vars{Identification rapide : $\omega_n$ = fr\'eq. naturelle (racine carr\'ee du terme constant), $\zeta$ = coeff. amor. (demi-coeff. lin. / $\omega_n$). Ex. : $s^2+s+10$ $\Rightarrow$ $\omega_n=\sqrt{10}{\approx}3.16$, $\zeta = 1/(2\sqrt{10}) \approx 0.158$.}
\end{tcolorbox}
